cmake_minimum_required(VERSION 3.16)
project(remani_adapter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(traj_opt REQUIRED)
find_package(plan_env REQUIRED)
find_package(mm_config REQUIRED)
find_package(mm_utils REQUIRED)
find_package(path_searching REQUIRED)
find_package(traj_utils REQUIRED)

add_library(remani_adapter src/planner.cpp)
target_include_directories(remani_adapter PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(remani_adapter Eigen3::Eigen)
ament_target_dependencies(remani_adapter rclcpp traj_opt plan_env mm_config mm_utils path_searching traj_utils)
install(TARGETS remani_adapter EXPORT export_remani_adapter LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_remani_adapter HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 rclcpp traj_opt plan_env mm_config mm_utils path_searching traj_utils)
add_executable(coupled_check src/coupled_check.cpp)
target_link_libraries(coupled_check remani_adapter)
ament_target_dependencies(coupled_check rclcpp traj_opt plan_env mm_config mm_utils path_searching traj_utils)
install(TARGETS coupled_check DESTINATION lib/${PROJECT_NAME})
ament_package()
