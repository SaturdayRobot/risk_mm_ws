cmake_minimum_required(VERSION 3.16)
project(path_searching)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(mm_utils REQUIRED)
find_package(plan_env REQUIRED)
find_package(mm_config REQUIRED)
find_package(traj_utils REQUIRED)
find_package(ompl REQUIRED)
include_directories(${OMPL_INCLUDE_DIRS})
link_libraries(${OMPL_LIBRARIES})
ament_export_include_directories(${OMPL_INCLUDE_DIRS})
ament_export_libraries(${OMPL_LIBRARIES})
ament_export_dependencies(ompl)

add_library(path_searching src/kino_astar.cpp src/rrt.cpp src/sample_mani_RRT.cpp)
target_include_directories(path_searching PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(path_searching SYSTEM PUBLIC ${OMPL_INCLUDE_DIRS})
target_link_libraries(path_searching Eigen3::Eigen)
ament_target_dependencies(path_searching rclcpp mm_utils plan_env mm_config traj_utils)
install(TARGETS path_searching EXPORT export_path_searching LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_path_searching HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 rclcpp mm_utils plan_env mm_config traj_utils)
ament_package()
