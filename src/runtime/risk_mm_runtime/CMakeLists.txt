cmake_minimum_required(VERSION 3.16)
project(risk_mm_runtime)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(remani_adapter REQUIRED)
find_package(plan_env REQUIRED)
find_package(mm_config REQUIRED)
find_package(mm_model REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(risk_mm_interfaces REQUIRED)
find_package(moveit_core REQUIRED)
find_package(geometric_shapes REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(tinyxml2_vendor REQUIRED)
find_package(TinyXML2 REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(world_scene src/world_scene.cpp src/task_goals.cpp)
target_include_directories(world_scene PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(world_scene Eigen3::Eigen yaml-cpp tinyxml2::tinyxml2 OpenSSL::Crypto)
ament_target_dependencies(world_scene sensor_msgs geometry_msgs gazebo_msgs)
target_compile_options(world_scene PRIVATE -Wall -Wextra -Wpedantic)
add_library(trajectory_validation src/validation.cpp src/time_scaling.cpp)
target_include_directories(trajectory_validation PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(trajectory_validation world_scene tinyxml2::tinyxml2)
ament_target_dependencies(trajectory_validation remani_adapter mm_model moveit_core geometric_shapes urdf srdfdom)
target_compile_options(trajectory_validation PRIVATE -Wall -Wextra -Wpedantic)
add_executable(coupled_probe src/coupled_probe.cpp)
target_link_libraries(coupled_probe world_scene trajectory_validation yaml-cpp)
ament_target_dependencies(coupled_probe rclcpp remani_adapter plan_env mm_config nav_msgs)
add_executable(scene_state_probe src/scene_state_probe.cpp)
target_link_libraries(scene_state_probe trajectory_validation yaml-cpp)
ament_target_dependencies(scene_state_probe rclcpp)
target_compile_options(scene_state_probe PRIVATE -Wall -Wextra -Wpedantic)
add_library(coupled_execution src/execution.cpp)
target_link_libraries(coupled_execution trajectory_validation)
ament_target_dependencies(coupled_execution mm_model)
add_library(coupled_patrol src/patrol.cpp)
target_link_libraries(coupled_patrol coupled_execution yaml-cpp)
add_library(coupled_runtime src/runtime.cpp)
target_link_libraries(coupled_runtime coupled_patrol yaml-cpp)
ament_target_dependencies(coupled_runtime rclcpp risk_mm_interfaces std_srvs plan_env mm_config nav_msgs)
foreach(target coupled_execution coupled_patrol coupled_runtime)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS world_scene trajectory_validation coupled_execution coupled_patrol coupled_runtime EXPORT export_world_scene LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(TARGETS coupled_probe scene_state_probe DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_world_scene HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp risk_mm_interfaces std_srvs plan_env mm_config nav_msgs sensor_msgs geometry_msgs gazebo_msgs std_msgs Eigen3 yaml-cpp tinyxml2_vendor TinyXML2 OpenSSL remani_adapter mm_model moveit_core geometric_shapes urdf srdfdom)
ament_package()
