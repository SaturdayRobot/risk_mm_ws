cmake_minimum_required(VERSION 3.16)
project(mm_bringup)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(mm_model REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TINYXML2 REQUIRED IMPORTED_TARGET tinyxml2)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(sim_model src/sim_model.cpp)
target_include_directories(sim_model PUBLIC include)
target_link_libraries(sim_model yaml-cpp PkgConfig::TINYXML2)
ament_target_dependencies(sim_model ament_index_cpp)
add_executable(prepare_sim src/prepare_sim.cpp)
target_link_libraries(prepare_sim sim_model)
add_executable(sim_control src/sim_control.cpp)
target_link_libraries(sim_control yaml-cpp PkgConfig::TINYXML2)
ament_target_dependencies(sim_control rclcpp ament_index_cpp geometry_msgs nav_msgs sensor_msgs std_msgs tf2_ros mm_model)
add_executable(sim_control_gui src/sim_control_gui.cpp)
target_link_libraries(sim_control_gui Qt5::Widgets PkgConfig::TINYXML2)
ament_target_dependencies(sim_control_gui rclcpp ament_index_cpp geometry_msgs sensor_msgs std_msgs)
install(TARGETS prepare_sim sim_control sim_control_gui DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch worlds DESTINATION share/${PROJECT_NAME})
ament_package()
